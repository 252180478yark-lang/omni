### 1.2 与路径 A 的复用 vs 升级

| 类别 | 复用 | 升级 |
|---|---|---|
| **scout-agent** | 框架 / runbook executor / 调度 | runbook YAML 模型加 platform_instance；多平台 session 管理 |
| **数据库** | mvp_* 数据通过迁移 | 新建 omni_* 三 schema |
| **前端** | 桥接组件（SaveToDecisionButton 等） | sidebar 重构、SKU 7 Tab、Dashboard |

| **现有模块** | knowledge / chat / roundtable / ad-review / content-studio 维持 | 接入 omni_core SKU 主数据 |

## 二、数据库设计

### 2.1 omni_core schema：资产层

```sql
CREATE SCHEMA IF NOT EXISTS omni_core;

-- ─── 店铺 ───
CREATE TABLE omni_core.shop (
    id              VARCHAR(64) PRIMARY KEY,         -- "shop-douyin-153xxx"
    platform        VARCHAR(32) NOT NULL,            -- douyin / tmall / jd
    platform_shop_id VARCHAR(128) NOT NULL,
    name            VARCHAR(200) NOT NULL,
    status          VARCHAR(32) DEFAULT 'active',
    metadata        JSONB DEFAULT '{}',
    created_at      TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(platform, platform_shop_id)
);

-- ─── SPU 主数据（你脑子里的"那一款产品"） ───
CREATE TABLE omni_core.sku (
    spu_id          VARCHAR(64) PRIMARY KEY,         -- "S001"
    name            VARCHAR(200) NOT NULL,
    category        VARCHAR(100),
    sku_attributes  JSONB DEFAULT '{}',              -- 物理属性：规格、原料、保质期
    push_tier       VARCHAR(32),                     -- main / growth / declining / new / archived
    notes           TEXT,
    metadata        JSONB DEFAULT '{}',
    created_at      TIMESTAMPTZ DEFAULT NOW(),
    updated_at      TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_sku_tier ON omni_core.sku(push_tier);

-- ─── SPU × 店铺 → 平台 SKU 链接 ───
CREATE TABLE omni_core.sku_platform_link (
    id                  BIGSERIAL PRIMARY KEY,
    spu_id              VARCHAR(64) NOT NULL REFERENCES omni_core.sku(spu_id) ON DELETE CASCADE,
    shop_id             VARCHAR(64) NOT NULL REFERENCES omni_core.shop(id) ON DELETE CASCADE,
    platform            VARCHAR(32) NOT NULL,
    platform_sku_id     VARCHAR(128) NOT NULL,
    platform_url        TEXT,
    status              VARCHAR(32) DEFAULT 'active',
    platform_specific   JSONB DEFAULT '{}',
    created_at          TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(shop_id, platform_sku_id)
);

CREATE INDEX idx_link_spu ON omni_core.sku_platform_link(spu_id);
CREATE INDEX idx_link_shop ON omni_core.sku_platform_link(shop_id);
CREATE INDEX idx_link_platform_sku ON omni_core.sku_platform_link(platform, platform_sku_id);

-- ─── 资产定义 ───
CREATE TABLE omni_core.asset (
    id                  BIGSERIAL PRIMARY KEY,
    platform_link_id    BIGINT NOT NULL REFERENCES omni_core.sku_platform_link(id) ON DELETE CASCADE,
    asset_type          VARCHAR(32) NOT NULL,        -- main_image / detail_page / title / price / video / audience_pack
    label               VARCHAR(100),
    shared_with         JSONB DEFAULT '[]',          -- 共享到哪些 platform
    is_active           BOOLEAN DEFAULT TRUE,
    created_at          TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_asset_link_type ON omni_core.asset(platform_link_id, asset_type);

-- ─── 资产版本 ───
CREATE TABLE omni_core.asset_version (
    id                  BIGSERIAL PRIMARY KEY,
    asset_id            BIGINT NOT NULL REFERENCES omni_core.asset(id) ON DELETE CASCADE,
    version_number      INT NOT NULL,                 -- v1, v2, v3...
    file_path           TEXT,
    thumbnail_path      TEXT,
    content             TEXT,                         -- 文字类资产（标题/价格/详情页文案）
    description         TEXT,
    activated_at        TIMESTAMPTZ NOT NULL,
    deactivated_at      TIMESTAMPTZ,
    activated_by        VARCHAR(32) DEFAULT 'owner',
    source_change_event_id BIGINT,                    -- 软引用 omni_decision.change_event
    metadata            JSONB DEFAULT '{}',
    created_at          TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(asset_id, version_number)
);

CREATE INDEX idx_version_asset_active ON omni_core.asset_version(asset_id, activated_at DESC);
```

### 2.2 omni_data schema：数据层

```sql
CREATE SCHEMA IF NOT EXISTS omni_data;

-- ─── 每日指标（必带 platform_link_id） ───
CREATE TABLE omni_data.daily_metric (
    id                  BIGSERIAL PRIMARY KEY,
    platform_link_id    BIGINT NOT NULL REFERENCES omni_core.sku_platform_link(id) ON DELETE CASCADE,
    date                DATE NOT NULL,
    metric_name         VARCHAR(64) NOT NULL,
    value               NUMERIC(20, 6),
    raw                 JSONB,
    source_runbook      VARCHAR(128),
    source_run_id       VARCHAR(64),
    created_at          TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(platform_link_id, date, metric_name)
);

CREATE INDEX idx_metric_link_date ON omni_data.daily_metric(platform_link_id, date DESC);
CREATE INDEX idx_metric_date_name ON omni_data.daily_metric(date DESC, metric_name);

-- ─── 异动事件 ───
CREATE TABLE omni_data.anomaly (
    id                  BIGSERIAL PRIMARY KEY,
    platform_link_id    BIGINT NOT NULL REFERENCES omni_core.sku_platform_link(id) ON DELETE CASCADE,
    detected_at         TIMESTAMPTZ DEFAULT NOW(),
    severity            VARCHAR(32),                  -- urgent / warning / positive
    rule_id             VARCHAR(64),
    metric_name         VARCHAR(64),
    description         TEXT,
    delta_pct           NUMERIC(8, 4),
    handled             BOOLEAN DEFAULT FALSE,
    handled_at          TIMESTAMPTZ,
    cross_platform_signature VARCHAR(64),             -- 跨平台聚合签名（同 SPU 跨平台异动聚合）
    created_at          TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_anomaly_unhandled ON omni_data.anomaly(handled, severity, detected_at DESC);
CREATE INDEX idx_anomaly_signature ON omni_data.anomaly(cross_platform_signature, detected_at DESC);

-- ─── 巡店执行记录 ───
CREATE TABLE omni_data.scout_run (
    id              VARCHAR(64) PRIMARY KEY,
    runbook_name    VARCHAR(128) NOT NULL,
    platform        VARCHAR(32) NOT NULL,
    shop_id         VARCHAR(64) REFERENCES omni_core.shop(id) ON DELETE SET NULL,
    started_at      TIMESTAMPTZ NOT NULL,
    ended_at        TIMESTAMPTZ,
    status          VARCHAR(32),
    error           TEXT,
    log_path        TEXT,
    metadata        JSONB DEFAULT '{}'
);

CREATE INDEX idx_run_name_time ON omni_data.scout_run(runbook_name, started_at DESC);

-- ─── 平台登录态 ───
CREATE TABLE omni_data.platform_session (
    shop_id         VARCHAR(64) PRIMARY KEY REFERENCES omni_core.shop(id),
    storage_path    TEXT NOT NULL,

    expires_at      TIMESTAMPTZ,
    health          VARCHAR(32) DEFAULT 'unknown',
    notes           TEXT,
    updated_at      TIMESTAMPTZ DEFAULT NOW()
);

-- ─── 客服会话（脱敏） ───
CREATE TABLE omni_data.customer_session (
    id                  BIGSERIAL PRIMARY KEY,
    platform_link_id    BIGINT REFERENCES omni_core.sku_platform_link(id) ON DELETE SET NULL,
    shop_id             VARCHAR(64) REFERENCES omni_core.shop(id) ON DELETE CASCADE,
    session_started_at  TIMESTAMPTZ,
    session_ended_at    TIMESTAMPTZ,
    customer_pseudonym  VARCHAR(64),                  -- 脱敏后的伪名
    transcript          TEXT,                          -- 全文（已脱敏）
    summary             TEXT,                          -- LLM 摘要
    pain_points         JSONB DEFAULT '[]',            -- 痛点 ID 列表
    sentiment           NUMERIC(3, 2),                 -- -1 ~ 1
    raw_metadata        JSONB DEFAULT '{}',
    created_at          TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_session_link_time ON omni_data.customer_session(platform_link_id, session_started_at DESC);

-- ─── 平台评价 ───
CREATE TABLE omni_data.review (
    id                  BIGSERIAL PRIMARY KEY,
    platform_link_id    BIGINT NOT NULL REFERENCES omni_core.sku_platform_link(id) ON DELETE CASCADE,
    review_text         TEXT,

CREATE INDEX idx_review_link_time ON omni_data.review(platform_link_id, posted_at DESC);

-- ─── 痛点分类（LLM 提取的标准化分类） ───
```

### 2.3 omni_decision schema：决策层

```sql
CREATE SCHEMA IF NOT EXISTS omni_decision;

-- ─── 动作日志 ───
CREATE TABLE omni_decision.change_event (
    id                      BIGSERIAL PRIMARY KEY,

    asset_version_after_id  BIGINT REFERENCES omni_core.asset_version(id) ON DELETE SET NULL,

    optimization_intent     VARCHAR(64),
    expected_kpis           JSONB DEFAULT '[]',
    executed_at             TIMESTAMPTZ NOT NULL,
    actor                   VARCHAR(32) DEFAULT 'owner',
    source_decision_log_id  BIGINT,
    verification_status     VARCHAR(32) DEFAULT 'pending',
    notes                   TEXT,
    cross_platform_group_id VARCHAR(64),               -- 同一动作同步多平台时的分组
    created_at              TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_event_link_time ON omni_decision.change_event(platform_link_id, executed_at DESC);
CREATE INDEX idx_event_verification ON omni_decision.change_event(verification_status, executed_at);
CREATE INDEX idx_event_cross_group ON omni_decision.change_event(cross_platform_group_id);

-- ─── AI 决策日志 ───
CREATE TABLE omni_decision.decision_log (
    id                      BIGSERIAL PRIMARY KEY,

    spu_id                  VARCHAR(64) REFERENCES omni_core.sku(spu_id) ON DELETE SET NULL,

    type                    VARCHAR(32),               -- diagnosis / suggestion / anomaly / experiment
    title                   VARCHAR(200) NOT NULL,
    summary                 TEXT,
    full_content            TEXT,
    status                  VARCHAR(32) DEFAULT 'pending',
    adopted_at              TIMESTAMPTZ,

    postponed_until         DATE,
    linked_change_event_ids JSONB DEFAULT '[]',        -- 多平台同时执行时关联多个 change_event

    verification_result     JSONB,
    meta                    JSONB DEFAULT '{}',
    created_at              TIMESTAMPTZ DEFAULT NOW(),
    updated_at              TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_decision_status_time ON omni_decision.decision_log(status, created_at DESC);
CREATE INDEX idx_decision_spu ON omni_decision.decision_log(spu_id, created_at DESC);

-- ─── 成功案例归档 ───
```

### 2.4 数据迁移：mvp_* → omni_*

文件：`migrations/path-b-001-from-mvp.sql`

```sql
-- ═════════════════════════════════════════════════════════════
-- 路径 A → 路径 B 数据迁移
-- 假设：路径 A 已有 mvp_sku（5 条）+ 各种 mvp_* 数据
-- 迁移策略：mvp_sku 拆为 sku（SPU）+ sku_platform_link（抖店 instance）
-- ═════════════════════════════════════════════════════════════

BEGIN;

-- 1. 创建抖店店铺
INSERT INTO omni_core.shop (id, platform, platform_shop_id, name)
SELECT
  'shop-douyin-' || COALESCE(douyin_shop_id, 'main'),
  'douyin',
  COALESCE(douyin_shop_id, 'main'),
  '抖店主店'
FROM (SELECT DISTINCT douyin_shop_id FROM mvp_sku) s
ON CONFLICT (platform, platform_shop_id) DO NOTHING;

-- 2. mvp_sku → omni_core.sku
INSERT INTO omni_core.sku (spu_id, name, category, push_tier, created_at, updated_at)
SELECT
  id, name, category, push_tier, created_at, updated_at
FROM mvp_sku
ON CONFLICT (spu_id) DO NOTHING;

-- 3. mvp_sku → omni_core.sku_platform_link（抖店 instance）
INSERT INTO omni_core.sku_platform_link
  (spu_id, shop_id, platform, platform_sku_id, platform_url, status, created_at)
SELECT
  s.id,
  'shop-douyin-' || COALESCE(s.douyin_shop_id, 'main'),
  'douyin',
  s.douyin_product_id,
  s.douyin_url,
  s.status,
  s.created_at
FROM mvp_sku s
ON CONFLICT (shop_id, platform_sku_id) DO NOTHING;

-- 4. mvp_daily_metric → omni_data.daily_metric
INSERT INTO omni_data.daily_metric
  (platform_link_id, date, metric_name, value, raw, source_runbook, source_run_id, created_at)
SELECT
  spl.id,
  m.date, m.metric_name, m.value, m.raw,
  m.source_runbook, m.source_run_id, m.created_at
FROM mvp_daily_metric m
JOIN omni_core.sku_platform_link spl ON spl.spu_id = m.sku_id AND spl.platform = 'douyin'
ON CONFLICT (platform_link_id, date, metric_name) DO NOTHING;

-- 5. mvp_change_event → omni_decision.change_event
INSERT INTO omni_decision.change_event
  (platform_link_id, asset_type, change_description, optimization_intent,
   expected_kpis, executed_at, actor, source_decision_log_id,
   verification_status, notes, created_at)
SELECT
  spl.id,
  e.asset_type, e.change_description, e.optimization_intent,
  e.expected_kpis, e.executed_at, e.actor, e.source_decision_log_id,
  e.verification_status, e.notes, e.created_at
FROM mvp_change_event e
JOIN omni_core.sku_platform_link spl ON spl.spu_id = e.sku_id AND spl.platform = 'douyin';

-- 6. mvp_decision_log → omni_decision.decision_log
INSERT INTO omni_decision.decision_log
  (source_module, source_run_id, spu_id, applies_to_platforms,
   type, title, summary, full_content, status,
   adopted_at, rejected_reason, postponed_until,
   verified_at, verification_result, meta, created_at, updated_at)
SELECT
  source_module, source_run_id, sku_id, '["douyin"]'::jsonb,
  type, title, summary, full_content, status,
  adopted_at, rejected_reason, postponed_until,
  verified_at, verification_result, meta, created_at, updated_at
FROM mvp_decision_log;

-- 7. mvp_anomaly → omni_data.anomaly
INSERT INTO omni_data.anomaly
  (platform_link_id, detected_at, severity, rule_id, metric_name,
   description, delta_pct, handled, handled_at, created_at)
SELECT
  spl.id,
  a.detected_at, a.severity, a.rule_id, a.metric_name,
  a.description, a.delta_pct, a.handled, a.handled_at, a.created_at
FROM mvp_anomaly a
JOIN omni_core.sku_platform_link spl ON spl.spu_id = a.sku_id AND spl.platform = 'douyin';

-- 8. mvp_runbook_run → omni_data.scout_run
INSERT INTO omni_data.scout_run
  (id, runbook_name, platform, shop_id, started_at, ended_at, status, error, log_path, metadata)
SELECT
  id, runbook_name, 'douyin',
  'shop-douyin-main',
  started_at, ended_at, status, error, log_path, metadata
FROM mvp_runbook_run;

-- 9. mvp_session → omni_data.platform_session
INSERT INTO omni_data.platform_session
  (shop_id, storage_path, last_login_at, expires_at, health, notes, updated_at)
SELECT
  'shop-douyin-main', storage_path, last_login_at, expires_at, health, notes, updated_at
FROM mvp_session WHERE platform = 'douyin'
ON CONFLICT (shop_id) DO UPDATE SET
  storage_path = EXCLUDED.storage_path,
  last_login_at = EXCLUDED.last_login_at;

-- 10. 重命名 mvp_* 表为 archived_mvp_*（保留 30 天后删除）
ALTER TABLE mvp_sku RENAME TO archived_mvp_sku;
ALTER TABLE mvp_daily_metric RENAME TO archived_mvp_daily_metric;
ALTER TABLE mvp_change_event RENAME TO archived_mvp_change_event;
ALTER TABLE mvp_verification RENAME TO archived_mvp_verification;
ALTER TABLE mvp_decision_log RENAME TO archived_mvp_decision_log;
ALTER TABLE mvp_anomaly RENAME TO archived_mvp_anomaly;
ALTER TABLE mvp_runbook_run RENAME TO archived_mvp_runbook_run;
ALTER TABLE mvp_session RENAME TO archived_mvp_session;

COMMIT;
```

### 2.5 索引策略

| 查询场景 | 关键索引 |
|---|---|
| 单 SKU 14 天数据 | `daily_metric (platform_link_id, date DESC)` |
| 跨平台 SPU 聚合 | `sku_platform_link (spu_id)` + 应用层 join |

```python
# anomaly_aggregator.py
"""
单 SPU 在多平台同一天触发多个异动 → 聚合为一个 cross_platform_signature。
工作台首页用这个 signature 把同 SPU 的异动合并展示。
"""

def make_signature(spu_id: str, date: date, severity: str) -> str:
    return f"{spu_id}:{date.isoformat()}:{severity}"

async def detect_anomalies_all_platforms(spu_id: str, date: date):
    links = await fetch_platform_links(spu_id)
    fired = []
    for link in links:
        anomalies = await detect_for_link(link.id, date)
        fired.extend(anomalies)
    # 标记跨平台签名
    if len({a["severity"] for a in fired}) > 0:
        sig = make_signature(spu_id, date, "mixed")
        await update_anomaly_signatures([a["id"] for a in fired], sig)
    return fired
```

## 七、关键代码骨架

### 7.1 跨平台聚合查询

```sql
-- 单 SPU 三平台 14 天 GMV 趋势
SELECT
  spl.platform,
  m.date,
  m.value AS gmv
FROM omni_data.daily_metric m
JOIN omni_core.sku_platform_link spl ON spl.id = m.platform_link_id
WHERE spl.spu_id = $1
  AND m.metric_name = 'gmv'
  AND m.date BETWEEN CURRENT_DATE - 14 AND CURRENT_DATE - 1
ORDER BY m.date, spl.platform;
```

### 7.2 命中率统计

```sql
-- 本周 AI 命中率
WITH this_week AS (
  SELECT * FROM omni_decision.decision_log
  WHERE created_at >= DATE_TRUNC('week', NOW())
    AND status IN ('verified', 'adopted', 'rejected', 'pending')
)
SELECT
  COUNT(*) AS total,
  COUNT(*) FILTER (WHERE status = 'adopted') AS adopted,
  COUNT(*) FILTER (WHERE verification_result->>'verdict' = 'positive') AS positive,
  ROUND(
    COUNT(*) FILTER (WHERE verification_result->>'verdict' = 'positive')::numeric /
    NULLIF(COUNT(*) FILTER (WHERE status = 'verified'), 0) * 100, 2
  ) AS hit_rate_pct
FROM this_week;
```
